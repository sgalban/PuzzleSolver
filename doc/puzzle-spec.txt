# This file describes the syntax of Puzzle Code in EBNF
# Note that while it's not explicitly defined, most non-terminals implicitly
# include a suffix of zero or more whitespace characters.

### EXPRESSIONS & NUMBERS

# Standard definition of natural number literals. Note that this language does
# not support negative numbers. Nonzero numbers cannot be written with leading
# zeroes.
<nonzeroDigit> := "1" | "2" | "3" | "4" | "5" | "6" | "7" | "8" | "9"
<digit> := "0" | nonzeroDigit
<number> := "0" | (nonzeroDigit, { digit })

# Expressions can be number literals, the application of a binary operation, a
# variable (applicable in only a few circumstances), or an expression wrapped in
# parentheses
<numExp> := number |
    (numExp, bop, numExp) |
    repeatParam |
    "(", numExp, ")"

# Ranges and number pairs are used in specific circumstances
<numPair> := "(", numExp, ",", numExp, ")"
<range> := numExp, "to", numExp

# Puzzle Code supports addition, subtraction, integer division, exponentiation,
# and the modulus operator.
# Division or modulus by 0, or subtraction that results in a negative, is an
# error.
<bop> := "+" | "-" | "*" | "//" | "^" | "%"

# A value representing the current iteration of a repeat constraint. They are
# represented by a number equal to the recursive depth of its corresponding
# repeat constraint. It's an error to use a repeatParam within a repeat
# constraint whose depth is less than the number used in its representation.
<repeatParam> := "$", number

### THE GRID

# The grid defines the puzzle. Note that the top left cell in the grid has the
# coordinate (0, 0). The first coordinate signifies the row of a cell, and the
# second signifies its column.
<grid> := "grid", numPair, "{", { constraints }, "}"

### CELLS

# Represents a group of 0 or more cells. This is is used to apply constraints to
# multiple cells at once. A single unary operator `~` can be used to invert the
# cell group.
<cells> := "~" (
    aCell |
    cellList |
    cellRowCol |
    cellRect |
    all |
    unconstrainedCells)

# A single cell from its coordinates on the grid. It's an error for these
# coordinates to lie outside the grid.
<aCell> := "cell", numPair

# A list of cells; the union of many cell groups. Essentially syntactic sugar
<cellList> := "cells", "[", [ cells, { ",", cells } ], "]"

# A group of horizontally or vertically adjacent cells. The row/column index is
# required, and optionally, the starting and ending columns/rows may be
# declared. If they are not, this represents the entire row/column of cells.
# It's an error to define boundaries that extend beyond the bounds of the grid.
<cellRowCol> := ("row" | "col"), numExp [ "from", numExp, "to", numExp ] 

# Represents a rectangle of cells whose top left cell is at the specified
# coordinate, and has the specified width and height. It's an error if it
# extends outside of the main grid.
<cellRect> := "subgrid", "(", numPair, ",", numExp, "," numExp, ")"

# Represents all the cells in the grid
<all> = "all"

# Represents all unconstrainted cells at the time this statement appears.
# In other words, it applies to all cells unaffected by constraints above it in
# the code.
unconstrainedCells := "unconstrained"

### CONSTRAINTS

# This defines a group of 0 or more constraints
<constraints> := cellInit | constraint | repeatConstraint

# A primitive constraint or a list of them, applied to a group of cells
<constraint> := (primitiveConstraint | constraintList), "in", cells

# A specialized constraint that sets the "given value" of a cell group.
# Essentially, this just constrains a cell to a single number.
<cellInit> := "init", cells, "with", numExp

# A single, primitive constraint
<primitiveConstraint> := "constraint",
    (uniqueConstraint |
    addsToConstraint |
    multToConstraint |
    greaterThanConstraint |
    lessThanConstraint |
    always |
    never)

# A constraint that is satisfied if all affected cells contain unqiue numbers
# within the given range. A constraint of this form is unsatisfiable if it
# contains more cells than the size of its range.
<uniqueConstraint> := "unique", "(", range, ")"

# A constraint that is satisfied if the numbers in all the affected cells add
# to the specified total
<addsToConstraint> := "addsTo", "(", numExp, ")"

# A constraint that is satisfied if the numbers in all the affected cells
# multiply to the specified total
<multiplyToConstraint> := "multTo", "(", numExp, ")"

# The union of several primitive constraints. Where order matters, they're
# applied from left to right
<constraintList> := "constraints", "[",
    [ primitiveConstraint, { ",", primitiveConstraint } ] "]"

# A constraint that is satisfied if all the affected cells contain a value
# greater than the cell at the specified coordinates.
<greaterThanConstraint> := "greaterThan", numPair

# A constraint that is satisfied if all the affected cells contain a value
# less than the cell at the specified coordinates.
<lessThanConstraint> := "lessThan", numPair

# A constraint that's always satisfied.
<always> = "always"

# A constraint that's never satisfied. Can be used for debugging purposes.
<never> = "never"

# Applies a constraint iteratively. The body of a repeat block can contain
# numerical expressions that use non-zero variables. In particular, if a
# a `repeat(x, y)` is nested within `n` `repeats`, its associated variable is
# `$n`, which increases from x to y (if x < y), or decreases from y to x (if
# y < x). This is syntactic sugar.
<repeatConstraint> := "repeat", numPair, "{", constraints, "}"
